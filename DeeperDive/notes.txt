/**
Go has 2 basic data structures for handling list data structs -

- Arrays - fixed length of things
- Slice - An array that can grow or shrink

Both slices and arrays must have a datatype- meaning we cannot store data of different types in an array or slice

**/

func newCard() string {
	return "Five of diamonds"
}

package main

import "fmt"

func main() {
	//here we are declaring cards as slice
	cards := []string{newCard(), newCard()}
	//Note - append function does not update the original slice , it returns a new slice to which we assign it back to the original
	cards = append(cards, "Six of spades")

	//Iterate of slice of cards and print the result
	/**
	range is a keyword that we use to iterate over the elements in an array or slice

	:= syntax is used because we are redeclaring and reassigning the values of i and card everytime we run the loop
	**/
	for i, card := range cards {
		fmt.Println(i, card)
	}

}

package main

import "fmt"

type deck []string

func newDeck() deck {
	cards := deck{}

	//in order to create the all 52 cobinations we are going to seperate the suits and values in 2 different sets and then use for loop twice to map the values to deck

	cardSuits := []string{"Spades", "Diamonds", "Hearts", "Clubs"}
	cardValues := []string{"Ace", "Two", "Three", "Four"}

	//after this we will setup two for loops nested inside one another to iterate through all the combinations

	//while looping if we have some variable which we are not going to use , we replace that variable with an _
	for _, suit := range cardSuits {
		for _, value := range cardValues {
			cards = append(cards, value+" of "+suit)
		}
	}
	return cards

}

func (d deck) print() {
	for i, card := range d {
		fmt.Println(i, card)
	}
}
